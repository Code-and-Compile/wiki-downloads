// Rich telemetry generator for ThingsBoard dashboards
// Based on the Code & Compile industrial motor demo.
let s = context.get('s') || {
    t: 0,
    baseTemp: 24,
    baseHum: 48,
    motorOn: true,
    speedRpm: 900,
    energyKWh: 0,
    runtimeSec: 0,
    cycleCount: 0,
    lastCycleT: 0,
    fault: false,
    faultCode: 0
};

s.t += 1;

function clamp(v, min, max) { return Math.max(min, Math.min(max, v)); }
function rand(min, max) { return min + Math.random() * (max - min); }
function chance(p) { return Math.random() < p; }

// Motor state machine.
if (chance(0.02)) s.motorOn = !s.motorOn;
if (chance(0.01) && !s.fault) {
    s.fault = true;
    s.faultCode = [101, 202, 303, 404][Math.floor(Math.random() * 4)];
}
if (s.fault && chance(0.15)) {
    s.fault = false;
    s.faultCode = 0;
}

// Speed ramping.
let targetRpm = s.motorOn && !s.fault ? rand(800, 1450) : 0;
s.speedRpm += (targetRpm - s.speedRpm) * 0.15;
s.speedRpm = clamp(s.speedRpm, 0, 1600);

// Environmental signals.
let dayWave = Math.sin(s.t / 120) * 1.5;
s.baseTemp += rand(-0.02, 0.02);
s.baseHum += rand(-0.05, 0.05);
let temperature = clamp(s.baseTemp + dayWave + rand(-0.2, 0.2), 18, 35);
let humidity = clamp(s.baseHum - dayWave + rand(-0.5, 0.5), 30, 70);

// Vibration rises with motor speed and faults.
let vibBase = (s.speedRpm / 1600) * 4.0;
let vibrationRms = vibBase + rand(0.1, 0.5);
if (s.fault) vibrationRms += rand(2, 5);
vibrationRms = clamp(vibrationRms, 0, 12);

// Electrical signals.
let currentA = s.motorOn && !s.fault
    ? clamp((s.speedRpm / 1600) * 7 + rand(0, 1), 0, 10)
    : rand(0, 0.3);
if (s.fault) currentA += rand(2, 4);

let voltageV = 230 + rand(-3, 3);
let powerW = clamp(voltageV * currentA * 0.85, 0, 4000);
s.energyKWh += (powerW / 1000) / 3600;

// Production signals.
if (s.motorOn && !s.fault) s.runtimeSec += 1;
if (s.motorOn && !s.fault && (s.t - s.lastCycleT) > rand(15, 35)) {
    s.cycleCount += 1;
    s.lastCycleT = s.t;
}
let cycleTimeSec = s.motorOn && !s.fault
    ? clamp(s.t - s.lastCycleT, 0, 60)
    : null;

// Quality and availability.
let rejectRate = s.fault ? rand(3, 12) : rand(0.3, 2.0);
let goodParts = Math.round(s.cycleCount * (1 - rejectRate / 100));
let availability = clamp((s.runtimeSec / s.t) * 100, 0, 100);

// Alarm state matches the dashboard thresholds.
let alarmLevel = 0;
if (temperature > 30 || vibrationRms > 7) alarmLevel = 1;
if (s.fault || vibrationRms > 9) alarmLevel = 2;

msg.payload = {
    temperature: Number(temperature.toFixed(2)),
    humidity: Number(humidity.toFixed(2)),
    motorOn: s.motorOn && !s.fault,
    speedRpm: Math.round(s.speedRpm),
    vibrationRms: Number(vibrationRms.toFixed(2)),
    currentA: Number(currentA.toFixed(2)),
    voltageV: Number(voltageV.toFixed(1)),
    powerW: Math.round(powerW),
    energyKWh: Number(s.energyKWh.toFixed(3)),
    cycleCount: s.cycleCount,
    cycleTimeSec: cycleTimeSec === null ? null : Number(cycleTimeSec.toFixed(1)),
    goodParts: goodParts,
    rejectRate: Number(rejectRate.toFixed(2)),
    availability: Number(availability.toFixed(1)),
    fault: s.fault,
    faultCode: s.faultCode,
    alarmLevel: alarmLevel
};

msg.topic = 'v1/devices/me/telemetry';
context.set('s', s);
node.status({
    fill: alarmLevel === 2 ? 'red' : alarmLevel === 1 ? 'yellow' : 'green',
    shape: 'dot',
    text: `${msg.payload.speedRpm} RPM | ${msg.payload.vibrationRms} mm/s`
});
return msg;
